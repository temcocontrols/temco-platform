// The build a documentation project leaves behind: `dist/docs`, the Markdown a reader can edit and the site publishes,
// with the reading order and every file a page points at copied beside it.
//
// Markdown and not HTML, on purpose. The site is a repository, and a page that is a `.md` file is a page anyone can fix
// through an Edit link; nothing here renders, because the reader's browser does that from this same file. The offline
// copies a reader downloads — the `.pdf` and the Windows help file — are built from these pages by the site, not here.
//
// `docs/help/registry.json` is the only source of truth: `ship` decides what is published, `title` becomes the page's
// front matter, and `order` is one number for the whole manual, so a chapter can fall between two pages of its parent —
// which is where the manual puts it.
//
// Temco Platform's Build Center runs this as the `docs:site` script and judges it by the files it leaves behind.

import { cpSync, mkdirSync, readdirSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';

const root = process.cwd();
const registry = readJson(join(root, 'docs', 'help', 'registry.json')) ?? { areas: {}, pages: {} };
const pkg = readJson(join(root, 'package.json')) ?? {};
const out = join(root, 'dist', 'docs');

const pages = Object.entries(registry.pages ?? {})
	.map(([id, page]) => ({ id, ...page }))
	.filter((page) => page.ship !== false)
	.sort((left, right) => (left.order ?? 0) - (right.order ?? 0) || left.id.localeCompare(right.id));

rmSync(out, { recursive: true, force: true });
mkdirSync(out, { recursive: true });

for (const page of pages) {
	const body = read(join(root, 'docs', page.id + '.md'));
	const file = join(out, page.id + '.md');
	mkdirSync(dirname(file), { recursive: true });
	writeFileSync(file, frontMatter(page) + withoutTitle(body, page.title ?? page.id));
}

// `00-index.md` and not `index.md`: the site serves a version from one file it knows by that name, and the `00-` keeps
// the front page first wherever the folder is listed by name.
writeFileSync(join(out, '00-index.md'), landing(pages));

// The reading order, as a file: a folder walk cannot know that "6.1 Graphic Screens" comes before "6.2 Programs", and
// the order lives in the registry rather than in the pages. This is the only place it is written down.
writeFileSync(
	join(out, 'nav.json'),
	JSON.stringify(
		{
			generated: new Date().toISOString().slice(0, 10),
			pages: pages.map((page) => ({ href: page.id + '.md', title: page.title ?? page.id, order: page.order ?? 0 }))
		},
		null,
		'\t'
	) + '\n'
);

// What the site needs to know about the project itself, beside the pages rather than inside them.
writeFileSync(
	join(out, 'site.json'),
	JSON.stringify(
		{
			title: String(pkg.name ?? 'Documentation'),
			section: 'documentation',
			...(typeof pkg.description === 'string' && pkg.description !== '' ? { summary: pkg.description } : {})
		},
		null,
		'\t'
	) + '\n'
);

// Everything a page points at — images, downloads, anything a topic carries — keeps the path the page gave it.
const assets = copyAssets(join(root, 'docs'), out);
console.log('docs: ' + pages.length + ' page(s) and ' + assets + ' asset(s) in ' + join('dist', 'docs'));

/** The front matter that makes a page findable: its title as a reader reads it, and where it sits in the manual. */
function frontMatter(page) {
	const lines = ['---', 'title: ' + JSON.stringify(page.title ?? page.id), 'order: ' + Number(page.order ?? 0)];
	if (typeof page.audience === 'string' && page.audience !== '') {
		lines.push('appliesTo: ' + JSON.stringify(page.audience));
	}
	lines.push('---', '');
	return lines.join('\n') + '\n';
}

/** A page's body without the title it repeats: the title is front matter now, so the first heading left is a section. */
function withoutTitle(body, title) {
	const lines = body.replace(/^\uFEFF/u, '').split(/\r?\n/u);
	while (lines.length > 0 && lines[0].trim() === '') {
		lines.shift();
	}
	// The site reads the title from front matter, so a body that opens with `#` prints it twice. A heading that says the
	// same thing goes; one that says something else becomes the page's first section instead of a second H1.
	const heading = /^#\s+(?<text>.*)$/u.exec(lines[0] ?? '');
	if (heading !== null) {
		lines.shift();
		while (lines.length > 0 && lines[0].trim() === '') {
			lines.shift();
		}
		const text = (heading.groups?.text ?? '').trim();
		if (text !== '' && !sameText(text, title)) {
			lines.unshift('## ' + text, '');
		}
	}
	return lines.join('\n').replace(/\s*$/u, '') + '\n';
}

/** Whether two headings are the same words, so the second one is a repeat: case and spacing are the writer's, not a difference. */
function sameText(left, right) {
	const flatten = (text) => String(text).replace(/\s+/gu, ' ').trim().toLowerCase();
	return flatten(left) === flatten(right);
}

/** The build's front page: what is here, in the order a reader would walk it. */
function landing(pages) {
	const name = String(pkg.name ?? 'Documentation');
	const lines = ['---', 'title: ' + JSON.stringify(name), 'order: 0', '---', ''];
	if (typeof pkg.description === 'string' && pkg.description !== '') {
		lines.push(pkg.description, '');
	}
	lines.push('## Contents', '');
	for (const page of pages) {
		const depth = String(page.id).split('/').length - 1;
		lines.push('  '.repeat(depth) + '- [' + (page.title ?? page.id) + '](' + page.id + '.md)');
	}
	lines.push('');
	return lines.join('\n');
}

function copyAssets(from, to) {
	let copied = 0;
	let entries = [];
	try {
		entries = readdirSync(from, { withFileTypes: true });
	} catch {
		return 0;
	}
	for (const entry of entries) {
		if (entry.name.startsWith('.')) {
			continue;
		}
		const source = join(from, entry.name);
		const target = join(to, entry.name);
		if (entry.isDirectory()) {
			// `help` is the registry, not a page; it has no business in a reader's build.
			if (entry.name === 'help') {
				continue;
			}
			copied += copyAssets(source, target);
			continue;
		}
		if (/\.md$/iu.test(entry.name)) {
			continue;
		}
		mkdirSync(dirname(target), { recursive: true });
		cpSync(source, target);
		copied += 1;
	}
	return copied;
}

function read(file) {
	try {
		return readFileSync(file, 'utf8');
	} catch {
		return '';
	}
}

function readJson(file) {
	try {
		return JSON.parse(readFileSync(file, 'utf8'));
	} catch {
		return undefined;
	}
}
